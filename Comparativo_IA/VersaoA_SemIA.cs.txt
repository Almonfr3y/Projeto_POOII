public Pedido Finalizar(int clienteId, int enderecoId, OpcaoEntrega opcaoEntrega, Pagamento pagamento, CarrinhoDeCompras carrinho)
{
    LojaDbContext db = new LojaDbContext();

    Pedido pedido = new Pedido();
    pedido.ClienteId = clienteId;
    pedido.EnderecoEntregaId = enderecoId;
    pedido.OpcaoEntregaId = opcaoEntrega.Id;
    pedido.Data = DateTime.Now;

    decimal peso = 0;
    int qtd = 0;
    foreach (PedidoItem i in carrinho.Itens)
    {
        Produto p = db.Produtos.Find(i.ProdutoId);
        if (p.Estoque < i.Quantidade)
            throw new Exception("Sem estoque do produto " + p.Nome);

        p.Estoque = p.Estoque - i.Quantidade;

        PedidoItem novo = new PedidoItem();
        novo.Produto = p;
        novo.ProdutoId = p.Id;
        novo.Quantidade = i.Quantidade;
        novo.PrecoUnitario = p.Preco;
        pedido.Itens.Add(novo);

        peso = peso + p.PesoKg * i.Quantidade;
        qtd = qtd + i.Quantidade;
    }

    // frete
    pedido.ValorFrete = opcaoEntrega.ValorBase + peso * opcaoEntrega.ValorPorKg;

    // desconto de 5% a partir de 5 peças
    if (qtd >= 5)
        pedido.ValorDesconto = pedido.CalcularSubtotal() * 0.05m;

    pagamento.Valor = pedido.CalcularSubtotal() + pedido.ValorFrete - pedido.ValorDesconto;
    pagamento.Processar();
    pedido.Pagamento = pagamento;
    pedido.Status = StatusPedido.Finalizado;

    db.Pedidos.Add(pedido);
    db.SaveChanges();

    carrinho.Limpar();
    return pedido;
}
